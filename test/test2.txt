#Q1. Hollow Square Star Pattern
#Write a Python program to print a hollow square pattern of stars for a given number `n`.

n=5
for i in range(n):
    if i == 0 or i == n - 1:
        print("* " * n)
    else:
        print("* " + "  " * (n - 2) + "*")


#Q2. Inverted Number Pyramid

#Write a Python program to print the following pattern.

num = 4
for i in range(num, 0, -1):
    for j in range(1, i + 1):
        print(j, end=" ")
    print()

#Q3. Check Whether a String is a Palindrome

#Write a Python program to check whether a given string is a palindrome (ignore case).
s="Madam"
s=s.lower()
if s==s[::-1]:
    print("Palindrome")
else:    
    print("Not a palindrome")


#Q4. Find the Most Frequent Character

#Write a Python program to find the most frequently occurring character in a string.


s = "success"
m = {}
for char in s:
    if char in m:
        m[char] += 1
    else:
        m[char] = 1
r= max(m, key=m.get)
print("Most frequent character:", r)


#Q5. Find Second Largest Element in a List
#Write a Python program to find the second largest number in a list without using built-in sort functions.

n = [12, 45, 23, 67, 45, 89]

largest = n[0]
s_largest = n[0]

for num in n:
    if num > largest:
        s_largest = largest
        largest = num
    elif num > s_largest and num != largest:
        s_largest = num

print("Second largest number:", s_largest)


#Q6. Rotate a List by K Positions

#Write a Python program to rotate a list to the right by `k` positions.
lst = [1, 2, 3, 4, 5]
k = 2

n = len(lst)
k = k % n
